#!/usr/bin/env python3
"""Run one frozen shop capture through the canonical MBP mirror runner."""
import fcntl
import hashlib
import importlib.util
import json
from pathlib import Path
import shlex
import subprocess
import sys
import tempfile

CANONICAL = Path('/Users/ArmanTarkhanian1/Library/Application Support/Anki2/addons21/ankiscape_fork')
CANDIDATE = Path('/Users/ArmanTarkhanian1/Library/Application Support/Anki2/ankiscape_worktrees/pool/.treehouse/ankiscape_fork-091389/5/ankiscape_fork')
HARNESS = Path('/tmp/ankiscape-shop-domain-split-capture-2026-09-05.py')
BINDINGS = Path('/tmp/ankiscape-shop-domain-split-bindings-2026-09-05.json')
HARNESS_SHA256 = 'deccfca3d8a376cc4b2358d1b1a9f7f9c225e0752d363c95f5ee5f0d9e71f4fb'
BINDINGS_SHA256 = '0473ab356907c7fad08aa7fffcfd1034f4fa358f84248ea77e9a7fc8888c231a'
RUNNER_SHA256 = '5de9414af35bb4688810d7d3f9b3331b205988b74a30706de6eca07af2de5e42'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest() if path.is_file() else None


def verify_local(expected, commit):
    if {name: sha(CANDIDATE / name) for name in expected} != expected:
        raise RuntimeError('Frozen source bindings changed; no remote commands launched')
    if subprocess.check_output(['git', '-C', str(CANDIDATE), 'rev-parse', 'HEAD'], text=True).strip() != commit:
        raise RuntimeError('Candidate base commit changed; no remote commands launched')
    if sha(HARNESS) != HARNESS_SHA256 or sha(BINDINGS) != BINDINGS_SHA256:
        raise RuntimeError('Frozen capture upload differs; no remote commands launched')
    if sha(CANONICAL / 'tools/run_remote_tests.py') != RUNNER_SHA256:
        raise RuntimeError('Canonical runner changed; no remote commands launched')


def main():
    if sha(BINDINGS) != BINDINGS_SHA256:
        raise RuntimeError('Bindings changed; no remote commands launched')
    bindings = json.loads(BINDINGS.read_text())
    expected = bindings['source_hashes']
    verify_local(expected, bindings['base_commit'])
    if sys.argv[1:] == ['--source-hash-canary']:
        wrong = {**expected, 'logic/shop_logic.py': '0' * 64}
        try:
            verify_local(wrong, bindings['base_commit'])
        except RuntimeError as error:
            if str(error) != 'Frozen source bindings changed; no remote commands launched':
                raise
            after = {name: sha(CANDIDATE / name) for name in expected}
            if after != expected:
                raise RuntimeError('Source changed during the local canary')
            print(json.dumps({
                'status': 'PASS', 'canary': str(error), 'remote_commands_launched': 0,
                'runner_imported': False, 'mirror_lock_acquired': False,
                'source_bindings_unchanged': True, 'source_file_count': len(expected),
                'deliberately_wrong_expected': wrong['logic/shop_logic.py'],
                'observed_sha256': after['logic/shop_logic.py'],
                'driver_sha256': sha(Path(__file__)),
            }), flush=True)
            return 0
        raise RuntimeError('Source-hash canary failed to stop; remote runner not entered')
    if sys.argv[1:]:
        raise RuntimeError('Unknown arguments; no remote commands launched')
    spec = importlib.util.spec_from_file_location('shop_domain_remote_runner', CANONICAL / 'tools/run_remote_tests.py')
    runner = importlib.util.module_from_spec(spec)
    sys.modules[spec.name] = runner
    spec.loader.exec_module(runner)
    summary = Path(tempfile.mkdtemp(prefix='shop-domain-split-driver-', dir='/tmp')) / 'capture.json'
    runner.LOCAL_MIRROR_LOCK_PATH.parent.mkdir(parents=True, exist_ok=True)
    with runner.LOCAL_MIRROR_LOCK_PATH.open('a+', encoding='utf-8') as lock:
        fcntl.flock(lock.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
        try:
            runner._reconcile_pending_run()
            runner._preflight('plain')
            runner._preflight('qt')
            verify_local(expected, bindings['base_commit'])
            uploads = {str(HARNESS): HARNESS_SHA256, str(BINDINGS): BINDINGS_SHA256}
            runner._run_with_retries(
                ('rsync', '-a', '-e', runner.RSYNC_SSH, *uploads, f'{runner.HOST}:/tmp/'),
                label='upload frozen shop capture and bindings')
            remote_hashes = runner._ssh(
                'shasum -a 256 ' + ' '.join(shlex.quote(path) for path in uploads),
                label='uploaded shop capture hash check', repeat_safe=True)
            observed = {line.split(maxsplit=1)[1].strip(): line.split(maxsplit=1)[0]
                        for line in remote_hashes.splitlines()}
            if observed != uploads:
                raise RuntimeError('Uploaded capture bindings differ; sync blocked')
            verify_local(expected, bindings['base_commit'])
            runner._sync(CANDIDATE)
            command = '/opt/homebrew/bin/python3 ' + shlex.quote(str(HARNESS))
            runner._remote_lane_command = lambda lane: command
            code, output = runner._stream_remote_run(
                'shop-domain-split-2026-09-05', timeout=3600, poll_interval=5)
            summary.write_text(json.dumps({'exit_code': code, 'output': output}, indent=2) + '\n')
            if code != 0:
                raise RuntimeError(f'Shop capture failed: {code}; local record: {summary}')
            report = Path(output.splitlines()[0])
            if report.parent != runner.REMOTE_ROOT / 'reports' or not report.name.startswith('shop-domain-split-'):
                raise RuntimeError(f'Unexpected report path: {report}')
            verdict = json.loads(runner._ssh(
                'cat ' + shlex.quote(str(report / 'verification.json')),
                label='shop terminal verification read', repeat_safe=True))
            if (verdict.get('status') != 'PASS' or verdict.get('errors') != []
                    or verdict.get('subprocess_count') != 29 or verdict.get('check_count') != 29):
                raise RuntimeError(f'Shop capture evidence did not pass: {verdict}')
            verify_local(expected, bindings['base_commit'])
            summary.write_text(json.dumps({
                'status': 'PASS', 'report': str(report), 'verification': verdict,
                'capture_sha256': HARNESS_SHA256, 'bindings_sha256': BINDINGS_SHA256,
                'runner_sha256': RUNNER_SHA256,
            }, indent=2) + '\n')
        finally:
            fcntl.flock(lock.fileno(), fcntl.LOCK_UN)
    print(json.dumps({'status': 'PASS', 'local_record': str(summary),
                      'report': str(report), 'verification': verdict}), flush=True)
    return 0


if __name__ == '__main__':
    raise SystemExit(main())
