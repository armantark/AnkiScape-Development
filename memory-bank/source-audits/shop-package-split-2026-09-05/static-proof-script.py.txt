"""Read-only structural proof for the shop split; imports no add-on code."""
import argparse
import ast
import builtins
import hashlib
import json
from pathlib import Path
import symtable

BASELINE_SHA = 'faf88588cc138c54456ffcf05476113666a9869591f72fd48a168118696dda7f'
DEFAULT_ROOT = '/Users/ArmanTarkhanian1/Library/Application Support/Anki2/ankiscape_worktrees/pool/.treehouse/ankiscape_fork-091389/5/ankiscape_fork'
parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('--root', type=Path, default=Path(DEFAULT_ROOT))
parser.add_argument('--baseline', type=Path, default=Path('/tmp/shop-domain-split-baseline-814b4d59.py'))
args = parser.parse_args()
root = args.root.resolve()
baseline_bytes = args.baseline.read_bytes()
assert hashlib.sha256(baseline_bytes).hexdigest() == BASELINE_SHA
baseline = baseline_bytes.decode()
old = ast.parse(baseline)
functions = {node.name: node for node in old.body if isinstance(node, ast.FunctionDef)}
state = [name for name, node in functions.items() if node.lineno < 268 or 689 <= node.lineno < 900 or node.lineno == 1536 or node.lineno >= 1578]
services = [name for name, node in functions.items() if 566 <= node.lineno <= 609 or 1019 <= node.lineno <= 1533 or node.lineno == 1540]
facade = [name for name in functions if name not in state + services]
owners = {'logic/shop_logic.py': facade, 'logic/shop/state.py': state, 'logic/shop/services.py': services}
assert tuple(map(len, (state, services, facade))) == (39, 18, 14)
constants = {node.targets[0].id: node for node in old.body if isinstance(node, ast.Assign)}
constant_owners = {
    'logic/shop_logic.py': [name for name in constants if name.startswith('_WILDERNESS_')],
    'logic/shop/state.py': ['MAX_SHOP_EVENT_HISTORY', '_CANONICAL_SHOP_ROW_NAMES'],
    'logic/shop/services.py': ['DECANTABLE_FOUR_DOSE_POTION_BASES', 'DECANTABLE_FLASK_POTION_BASES'],
}
public_moved = [name for name in state + services if not name.startswith('_')]
original_imports = [node for node in old.body if isinstance(node, (ast.Import, ast.ImportFrom))]
inputs = {args.baseline.resolve(): baseline_bytes}
report = {'baseline_sha256': BASELINE_SHA, 'files': {}, 'functions': {}, 'constants': {}}
replacements = {}


def dumped(node):
    return ast.dump(node, include_attributes=False)


def exact_source(source, node):
    return ''.join(source.splitlines(keepends=True)[node.lineno - 1:node.end_lineno])


def referenced_globals(table):
    names = {symbol.get_name() for symbol in table.get_symbols() if symbol.is_global() and symbol.is_referenced()}
    for child in table.get_children():
        names.update(referenced_globals(child))
    return names


for relative, expected_functions in owners.items():
    path = root / relative
    raw = path.read_bytes()
    inputs[path] = raw
    source = raw.decode()
    tree = ast.parse(source, filename=str(path))
    line_count = len(source.splitlines())
    assert line_count < 1000, (relative, line_count)
    actual_functions = [node.name for node in tree.body if isinstance(node, ast.FunctionDef)]
    assert actual_functions == expected_functions, (relative, actual_functions)
    actual_constants = []
    metadata_names = []
    for index, node in enumerate(tree.body):
        if isinstance(node, ast.FunctionDef):
            before = functions[node.name]
            assert dumped(node) == dumped(before), node.name
            moved_source = exact_source(source, node)
            assert moved_source == exact_source(baseline, before), node.name
            replacements[before.lineno] = (before.end_lineno, moved_source)
            report['functions'][node.name] = {
                'owner': relative,
                'line': node.lineno,
                'baseline_line': before.lineno,
                'source_sha256': hashlib.sha256(moved_source.encode()).hexdigest(),
            }
        elif isinstance(node, ast.Assign):
            assert len(node.targets) == 1
            target = node.targets[0]
            if isinstance(target, ast.Name):
                assert target.id in constant_owners[relative], (relative, target.id)
                assert dumped(node) == dumped(constants[target.id]), target.id
                assert exact_source(source, node) == exact_source(baseline, constants[target.id]), target.id
                actual_constants.append(target.id)
                before = constants[target.id]
                replacements[before.lineno] = (before.end_lineno, exact_source(source, node))
                report['constants'][target.id] = {'owner': relative, 'line': node.lineno}
            else:
                assert relative == 'logic/shop_logic.py'
                assert isinstance(target, ast.Attribute) and target.attr == '__module__'
                assert isinstance(target.value, ast.Name) and target.value.id in public_moved
                assert isinstance(node.value, ast.Name) and node.value.id == '__name__'
                metadata_names.append(target.value.id)
        elif isinstance(node, (ast.Import, ast.ImportFrom)):
            assert all(alias.name != '*' for alias in node.names)
        else:
            assert index == 0 and isinstance(node, ast.Expr) and isinstance(node.value, ast.Constant) and isinstance(node.value.value, str), (relative, dumped(node))
    assert actual_constants == constant_owners[relative], relative
    assert metadata_names == (public_moved if relative == 'logic/shop_logic.py' else []), relative
    symbols = symtable.symtable(source, str(path), 'exec')
    available = {symbol.get_name() for symbol in symbols.get_symbols() if symbol.is_assigned() or symbol.is_imported() or symbol.is_namespace()}
    assert not (referenced_globals(symbols) - available - set(vars(builtins))), relative
    imports = [node for node in tree.body if isinstance(node, (ast.Import, ast.ImportFrom))]
    if relative == 'logic/shop_logic.py':
        assert [dumped(node) for node in imports[:len(original_imports)]] == [dumped(node) for node in original_imports]
        extra = imports[len(original_imports):]
        assert [(node.level, node.module) for node in extra] == [(1, 'shop.state'), (1, 'shop.services')]
        for node, child in zip(extra, ('logic/shop/state.py', 'logic/shop/services.py')):
            assert [(alias.name, alias.asname) for alias in node.names] == [(name, None) for name in constant_owners[child] + owners[child]]
    else:
        assert all(not isinstance(node, ast.Import) for node in imports if any(alias.name.endswith('shop_logic') for alias in node.names))
        assert all(not isinstance(node, ast.ImportFrom) or node.module != 'shop_logic' for node in imports)
    report['files'][relative] = {'sha256': hashlib.sha256(raw).hexdigest(), 'lines': line_count, 'functions': len(actual_functions)}
assert set(report['functions']) == set(functions)
assert set(report['constants']) == set(constants)
reconstructed = baseline.splitlines(keepends=True)
for first, (last, replacement) in sorted(replacements.items(), reverse=True):
    reconstructed[first - 1:last] = replacement.splitlines(keepends=True)
reconstructed_bytes = ''.join(reconstructed).encode()
assert reconstructed_bytes == baseline_bytes
report['reconstructed_baseline_sha256'] = hashlib.sha256(reconstructed_bytes).hexdigest()

proof_test_path = root / 'tests/test_quest_action_proofs.py'
proof_test = proof_test_path.read_bytes()
inputs[proof_test_path] = proof_test
proof_tree = ast.parse(proof_test)
method = next(node for node in ast.walk(proof_tree) if isinstance(node, ast.FunctionDef) and node.name == 'test_production_proof_writers_are_genuine_action_seams')
expected_sites = ast.literal_eval(method.body[1].value.args[1])
expected_mentions = ast.literal_eval(method.body[2].value.args[1])
recorders = {'record_runtime_quest_action_proofs_pure', 'record_quest_stage_action_proofs_pure', 'retract_shop_event_quest_action_proofs_pure'}
actual_sites = set()
actual_mentions = {}
paths = [root / '__init__.py']
for directory in ('core', 'logic', 'logic_pure', 'runtime', 'ui', 'data', 'generated'):
    paths.extend(sorted((root / directory).rglob('*.py')))
for path in paths:
    raw = path.read_bytes()
    inputs[path] = raw
    source = raw.decode()
    count = source.count('quest_action_proofs')
    if not count:
        continue
    relative = str(path.relative_to(root))
    actual_mentions[relative] = count
    tree = ast.parse(source)
    function_names = {}
    for function in ast.walk(tree):
        if isinstance(function, (ast.FunctionDef, ast.AsyncFunctionDef)):
            for node in ast.walk(function):
                function_names[node] = function.name
    for node in ast.walk(tree):
        if isinstance(node, ast.Call) and isinstance(node.func, ast.Name) and node.func.id in recorders:
            actual_sites.add((relative, function_names.get(node, '<module>'), node.func.id))
assert actual_sites == expected_sites, (actual_sites, expected_sites)
assert actual_mentions == expected_mentions, (actual_mentions, expected_mentions)
report['proof_writer_sites'] = sorted(actual_sites)
report['proof_writer_mentions'] = actual_mentions
report['proof_test_sha256'] = hashlib.sha256(proof_test).hexdigest()
assert not (root / 'logic/shop_state.py').exists()
assert not (root / 'logic/shop_services.py').exists()
assert (root / 'logic/shop/__init__.py').read_bytes() == b''
for relative in ('logic/shop/__init__.py', 'tests/shops/__init__.py', 'tests/shops/test_module_boundaries.py', 'tests/test_module_surface.py'):
    raw = (root / relative).read_bytes()
    inputs[root / relative] = raw
    ast.parse(raw)
    report['files'][relative] = {'sha256': hashlib.sha256(raw).hexdigest(), 'lines': len(raw.splitlines())}
assert all(path.read_bytes() == raw for path, raw in inputs.items())
report['unchanged_input_files'] = len(inputs)
report['status'] = 'PASS_STATIC_ONLY'
print(json.dumps(report, indent=2, sort_keys=True))
