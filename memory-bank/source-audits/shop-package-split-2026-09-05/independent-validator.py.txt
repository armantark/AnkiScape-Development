from pathlib import Path
from collections import Counter
from datetime import datetime, timezone
import hashlib
import json

ROOT = Path('/tmp/shop-package-split-oyfvtlm2')
BINDING_PATH = Path('/tmp/ankiscape-shop-package-split-bindings-2026-09-05.json')

def digest(raw):
    return hashlib.sha256(raw).hexdigest()

def read(name):
    return json.loads((ROOT / name).read_text())

assert digest(BINDING_PATH.read_bytes()) == 'dee9fc31ed5e316ff7d18cf80ce70ae1191c611bc414b25e07f4521d197686ef'
bindings = json.loads(BINDING_PATH.read_text())
expected = bindings['source_hashes']
assert len(expected) == 38
capture = read('capture.json')
assert capture['root'] == '/Users/armantarkhanian/ankiscape-remote-test-host/ankiscape_fork'
assert capture['base_commit'] == bindings['base_commit']
assert capture['script_sha256'] == '50f87266123b51ac9250af9578b9c0397a84bec66df179c7eb1d52c362851f33'
assert capture['bindings_sha256'] == digest(BINDING_PATH.read_bytes())
for key in ('focus_ids', 'surface_test_ids', 'added_ids'):
    assert capture[key] == bindings[key], key
assert capture['required_payloads'] == list(bindings['baseline_surfaces'])
assert read('source-hashes.json') == {'expected': expected, 'actual': expected}
assert read('source-hashes-after.json') == expected
verdict = read('verification.json')
assert verdict['status'] == 'PASS' and verdict['errors'] == []
assert verdict['subprocess_count'] == verdict['check_count'] == 29
results, checks = read('results.json'), read('checks.json')
assert len(results) == len(checks) == 29
streams, test_counts, discovery_counts, payloads = {}, {}, {}, {}
command_keys = []
for lane in ('plain', 'qt'):
    command_keys += [(lane, mode) for mode in ('discovery', 'focus', 'module-surface-tests')]
    command_keys += [(lane, module.replace('/', '_')) for module in bindings['modules'] + (['__init__.py'] if lane == 'qt' else [])]
assert [(r['lane'],r['mode']) for r in results] == command_keys
assert [(r['lane'],r['mode']) for r in checks] == command_keys
for row, check in zip(results, checks):
    lane, mode = row['lane'], row['mode']
    assert row['status'] == 'finished' and row['exit_code'] == 0 and check['status'] == 'PASS'
    assert row['source_hashes_before'] == row['source_hashes_after'] == expected
    assert row['cwd'] == capture['root']
    interpreter = '/opt/homebrew/bin/python3' if lane == 'plain' else '/Users/armantarkhanian/ankiscape-remote-test-host/.venv-qt/bin/python'
    if mode == 'discovery':
        args = ['run_tests.py', '--list', '--max-results', '10000']
    elif mode == 'focus':
        args = ['run_tests.py', *bindings['focus_ids'], '--verbosity', '2']
    elif mode == 'module-surface-tests':
        args = ['run_tests.py', 'test_module_surface.', '--verbosity', '2']
    else:
        candidates = [m for m in bindings['modules'] + ['__init__.py'] if m.replace('/', '_') == mode]
        assert len(candidates) == 1
        args = ['tools/module_surface.py', candidates[0]]
    assert row['argv'] == [interpreter, *args]
    assert row['started_utc'] <= row['finished_utc']
    for stream in ('stdout', 'stderr'):
        name = f'{lane}-{mode}.{stream}'
        assert row[stream] == name and name not in streams
        raw = (ROOT / name).read_bytes()
        assert len(raw) == row[stream+'_bytes'] and digest(raw) == row[stream+'_sha256']
        streams[name] = {'bytes':len(raw), 'sha256':digest(raw)}
    stdout = (ROOT / row['stdout']).read_bytes()
    stderr = (ROOT / row['stderr']).read_text()
    if mode == 'discovery':
        baseline = (Path('/tmp') / bindings['baseline_discovery_directory'] / (lane+'-discovery.stdout')).read_bytes()
        assert digest(baseline) == bindings['baseline_discovery_sha256']
        old = baseline.decode().splitlines(); ids = stdout.decode().splitlines()
        assert len(old) == 5567 and len(ids) == len(set(ids)) == 5568 and stderr == ''
        assert Counter(t for t in ids if t in bindings['added_ids']) == Counter(bindings['added_ids'])
        assert [t for t in ids if t not in bindings['added_ids']] == old
        assert check['count'] == 5568 and check['added_ids'] == bindings['added_ids'] and check['existing_order_preserved'] is True
        disc = read('discovery.json')[lane]
        assert disc == {'count':5568, 'ids':ids, 'stdout_sha256':digest(stdout)}
        discovery_counts[lane] = {'baseline':len(old), 'candidate':len(ids),'sha256':digest(stdout)}
    elif mode in ('focus', 'module-surface-tests'):
        wanted = bindings['focus_ids'] if mode == 'focus' else bindings['surface_test_ids']
        outcomes = check['outcomes']; ids = [r['id'] for r in outcomes]
        assert Counter(ids) == Counter(wanted) and len(set(ids)) == len(wanted)
        assert stdout == b'' and check['count'] == len(wanted)
        rendered = ''
        skips = 0
        for test in outcomes:
            if test['status'] != 'ok':
                assert mode == 'module-surface-tests'
                assert test['id'] == 'test_module_surface.ModuleSurfaceTest.test_root_initializer_is_reported_as_unavailable_without_aqt'
                assert test['status'] == "skipped 'root initializer requires aqt; use the offscreen Qt interpreter'"
                skips += 1
            rendered += test['id'].rsplit('.',1)[-1] + ' (' + test['id'] + ')'
            if test['description'] is not None:
                rendered += '\n' + test['description']
            rendered += ' ... ' + test['status'] + '\n'
        assert skips == check['skips']
        ending = 'OK' + (f' (skipped={skips})' if skips else '')
        rendered += '\n' + '-'*70 + f"\nRan {len(wanted)} tests in {check['seconds']}s\n\n{ending}\n"
        assert stderr == rendered, (lane,mode,'raw unittest output differs from verified exact result')
        test_counts[lane+'-'+mode] = {'tests':len(ids),'skips':skips,'seconds':check['seconds']}
    else:
        original = (Path('/tmp') / bindings['baseline_surfaces_directory'] / row['stdout']).read_bytes()
        assert stdout == original
        body, footer = stdout.rsplit(b'MODULE_SURFACE_SHA256: ',1)
        assert body.startswith(b'MODULE_SURFACE module=') and footer == (digest(body)+'\n').encode()
        measured = {'bytes':len(stdout),'file_sha256':digest(stdout),'body_sha256':digest(body)}
        assert measured == bindings['baseline_surfaces'][row['stdout']]
        assert check['byte_identical'] is True and all(check[k] == v for k,v in measured.items())
        payloads[row['stdout']] = measured
assert len(streams) == 58 and len(payloads) == 23
assert set(p.name for p in ROOT.glob('*.stdout')) | set(p.name for p in ROOT.glob('*.stderr')) == set(streams)
base = Path('/tmp') / bindings['baseline_discovery_directory']
base_hashes = {name:digest((base/name).read_bytes()) for name in bindings['baseline_report_hashes']}
assert base_hashes == bindings['baseline_report_hashes']
base_integrity = {'report_hashes':base_hashes, 'surfaces':payloads,
                  'discovery_sha256':bindings['baseline_discovery_sha256'],'discovery_count':5567}
assert read('baseline-integrity.json') == read('baseline-integrity-after.json') == base_integrity
files = {p.name:{'bytes':p.stat().st_size,'sha256':digest(p.read_bytes())} for p in sorted(ROOT.iterdir()) if p.is_file()}
assert len(files) == 67 and len(list(ROOT.iterdir())) == 67
report = {'status':'PASS_INDEPENDENT_LOCAL_VALIDATION','verified_utc':datetime.now(timezone.utc).isoformat(),
          'report':str(ROOT),'remote_report':capture['root'].replace('/ankiscape_fork','/reports/shop-package-split-oyfvtlm2'),
          'capture_session':'73051','copy_session':'47391','source_file_count':38,'finished_exit0_jobs':29,
          'raw_stream_count':58,'whole_payloads_byte_identical':23,'test_counts':test_counts,
          'discovery':discovery_counts,'raw_file_count':67,'raw_total_bytes':sum(r['bytes'] for r in files.values()),
          'files':files,'validator_sha256':digest(Path(__file__).read_bytes()),
          'limitations':'This validates retained focused evidence only; no full gate, commit, or merged-tree acceptance is claimed. No source worktree or mirror was edited.'}
p=Path('/tmp/shop-package-split-oyfvtlm2-independent-validation.json');p.write_text(json.dumps(report,indent=2)+'\n')
print(json.dumps({k:v for k,v in report.items() if k!='files'},indent=2))
print('VALIDATION_SHA256',digest(p.read_bytes()))
